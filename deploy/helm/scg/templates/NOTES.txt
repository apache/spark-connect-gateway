  kubectl -n {{ .Release.Namespace }} port-forward svc/{{ include "scg.fullname" . }} {{ .Values.service.adminPort }}:{{ .Values.service.adminPort }} &
  curl http://localhost:{{ .Values.service.adminPort }}/metrics
